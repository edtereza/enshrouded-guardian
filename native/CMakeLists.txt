cmake_minimum_required(VERSION 3.28)
project(guardian_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
include(FetchContent)
set(SAFETYHOOK_FETCH_ZYDIS ON CACHE BOOL "" FORCE)
FetchContent_Declare(safetyhook GIT_REPOSITORY https://github.com/cursey/safetyhook.git GIT_TAG v0.7.0)
FetchContent_MakeAvailable(safetyhook)
add_library(guardian_native STATIC hardcore.cpp log.cpp)
target_link_libraries(guardian_native PRIVATE safetyhook::safetyhook)
target_compile_definitions(guardian_native PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
enable_testing()
add_executable(guardian_native_test tests/hardcore_test.cpp tests/host.cpp hardcore.cpp log.cpp)
target_include_directories(guardian_native_test PRIVATE .)
target_link_libraries(guardian_native_test PRIVATE safetyhook::safetyhook)
target_compile_definitions(guardian_native_test PRIVATE HARDCORE_TEST WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
add_test(NAME guardian_native COMMAND guardian_native_test)
add_executable(guardian_native_probe tests/probe.cpp tests/host.cpp hardcore.cpp log.cpp)
target_include_directories(guardian_native_probe PRIVATE .)
target_link_libraries(guardian_native_probe PRIVATE safetyhook::safetyhook)
target_compile_definitions(guardian_native_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)

add_executable(guardian_loot_test tests/loot_test.cpp)
target_include_directories(guardian_loot_test PRIVATE .)
add_test(NAME guardian_loot COMMAND guardian_loot_test)

add_executable(guardian_dll_test tests/dll_test.cpp)
target_compile_definitions(guardian_dll_test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE GUARDIAN_VERSION="${GUARDIAN_VERSION}")
